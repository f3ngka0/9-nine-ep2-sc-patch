# コンフィグ画面ヘルプテキスト

#------------------------------------------------------------------------------
# 共通

page0	page1	page2	page3	page4	page5
	切换设置分类标签页。

init
	初始化设置。
title
	返回标题画面。
exit
	退出游戏。
back
	返回上一个画面。
quit
	返回鉴赏画面。

#------------------------------------------------------------------------------
# 言語切り替え

lang_ui	lang_ui_l	lang_ui_r
	更改界面（UI）使用的语言。

lang_main	lang_man_l	lang_main_r
	更改游戏正文的主语言。

lang_sub	lang_sub_l	lang_sub_r
	更改游戏正文的次语言。	与主语言相同或设为关闭时不会显示。


#------------------------------------------------------------------------------
# システム

window	fullscreen
	切换画面显示尺寸。

noeffect_off	noeffect_on
	切换是否开启画面特效。

bganim_on	bganim_off
	切换是否显示 H 场景动画。

rtextcol_on	rtextcol_off
	切换已读文本是否改变颜色。

rselcol_on	rselcol_off
	切换已选过的选项是否改变颜色。

curmove_on	curmove_off
	开启/关闭出现选项或确认对话框时，鼠标光标自动移动的功能。

ezsave_on	ezsave_off
	开启/关闭位于游戏画面右侧的简易存取档功能。

cdown_on	cdown_off
	切换是否显示射精倒计时提示。

hsel_s_in	hsel_s_out	hsel_s_ask	hsel_f_in	hsel_f_out	hsel_f_ask
	切换是否固定射精位置选项，或设为每次询问。

danmen_on	danmen_off
	切换是否显示断面图特写画面。

facemode_on	facemode_off
	切换是否显示角色表情头像。（因演出需要，设为关闭时部分场景仍会显示。）

#chareye_on	chareye_off
#	切换角色是否眨眼。
chareye_on	chareye_off
	切换是否开启索菲蒂娅的手部动作演出。

fcamera_on	fcamera_off
	切换是否开启自由视角功能。（当鼠标光标变为“＋”时拖拽即可启用。）

#------------------------------------------------------------------------------
# メッセージ

textspeed
	设置常规模式下的文本显示速度。

atextspeed
	设置自动模式下的文本显示速度。

autospeed
	设置自动模式下的自动推进速度。

winopac
	设置文本框背景图片的不透明度。

fshadow_on	fshadow_off
	开启/关闭文本的阴影效果。

skipread	skipall
	切换快进（略过）模式的有效范围。

voicecut_on	voicecut_off
	切换推进至下段对话时，是否中断正在播放的语音。

autovwait_on	autovwait_off
	设置自动模式下是否等待当前语音播放完毕再推进。

sample_normal	sample_auto	sample_off
	切换测试文本示例的显示模式。

wintype_1	wintype_2	sample_wintype
	选择对话框的样式类型。

#------------------------------------------------------------------------------
# サウンド

wavevolume
	调整主音量。
wave_on	wave_off
	开启/关闭所有声音播放。

bgmvolume
	调整 BGM 音量。
bgm_on	bgm_off
	开启/关闭 BGM 播放。

downvolume
	设置语音播放时的 BGM 音量（不可高于 BGM 原音量）。
bgmdown_on	bgmdown_off
	开启/关闭语音播放时降低 BGM 音量的功能。

sevolume
	调整音效音量。
se_on	se_off
	开启/关闭音效播放。
setest
	试听音效。

hsevolume
	调整 H 场景特殊音效音量。
hse_on	hse_off
	开启/关闭 H 场景特殊音效播放。
hsetest
	试听 H 场景特殊音效。

bgvvolume
	调整 H 场景背景语音（BGV）音量。
bgv_on	bgv_off
	开启/关闭 H 场景背景语音（BGV）播放。
bgvtest
	试听 H 场景背景语音（BGV）。再次点击即可停止。

voicevolume
	调整所有角色的全局语音音量。
voice_on	voice_off
	开启/关闭语音播放。

#chvoicearea
#	勾选角色即可单独设置该角色的语音音量。

#autostopse_on	autostopse_off
#	设置解除略过模式时的音效播放。

#skipstopse_on	skipstopse_off
#	设置解除自动模式时的音效播放。

stopse_on	stopse_off
	设置解除略过/自动模式时是否播放提示音效。

bgmtitle_on	bgmtitle_off
	设置是否显示游戏中当前播放的 BGM 曲名。

sysvovolume
	调整系统语音音量。

sysvoice_on	sysvoice_off
	开启/关闭系统语音播放。

sysvotarget	sysvo_up,	sysvo_down
	切换系统语音的担当角色。

sysvotest
	试听系统语音。

sysvo_clrvo
	${SystemVoice.chartype=='5人'?"仅将已通关角色切换为通关后语音模式":"将系统语音切换为通关后语音模式"}。
